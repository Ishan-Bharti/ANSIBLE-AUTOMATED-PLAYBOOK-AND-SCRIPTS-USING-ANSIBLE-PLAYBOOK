  pre_tasks:
    - name: Create report directory on control node
      run_once: true
      delegate_to: localhost
      file:
        path: "{{ report_dir }}"
        state: directory
        mode: '0755'

  tasks:
    - name: Set connectivity list for this host
      set_fact:
        connectivity_targets: "{{ server_ports[host_key_map[inventory_hostname]] }}"
      when: inventory_hostname in host_key_map

    - name: Check connectivity for each port
      shell: nc -z -w 5 {{ item.host }} {{ item.port }}
      register: nc_result
      loop: "{{ connectivity_targets }}"
      ignore_errors: yes
      when: connectivity_targets is defined

    - name: Build host report string
      set_fact:
        host_report: |
          ==========================================
          Server: {{ inventory_hostname }}
          Timestamp: {{ ansible_date_time.iso8601 if ansible_date_time is defined else lookup('pipe','date -Iseconds') }}
          ==========================================
          {% for r in nc_result.results %}
          {{ r.item.name }} ({{ r.item.host }}:{{ r.item.port }}) : {{ 'CONNECTED' if r.rc == 0 else 'NOT REACHABLE' }}
          {% endfor %}

    - name: Write host report to a separate file on control node
      delegate_to: localhost
      copy:
        content: "{{ host_report }}"
        dest: "{{ report_dir }}/port_connectivity_{{ inventory_hostname }}.txt"
        force: yes
